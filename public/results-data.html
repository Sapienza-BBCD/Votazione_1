app.get("/results-data", (req, res) => { db.all(` SELECT percorso, titolo, COUNT(*) as votes FROM votes GROUP BY percorso, titolo `, (err, rows) => { if (err) { return res.json({ error: err.message }); } const result = {}; for (const r of rows) { // disciplina = parte prima del " - " const disciplina = r.percorso.split(" - ")[0].trim(); if (!result[disciplina]) { result[disciplina] = []; } result[disciplina].push({ id: r.titolo, // puoi anche rinominarlo in "progetto" votes: r.votes }); } // ordina ogni disciplina per voti decrescenti for (const d in result) { result[d].sort((a, b) => b.votes - a.votes); } res.json(result); }); });
